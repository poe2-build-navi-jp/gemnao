import {runFixedQa} from './ops/private-diagnostic-qa/driver.mjs';import {qaAccessWindow,qaAccessVerifier} from './ops/private-diagnostic-qa/existing-qa-policy.mjs';import {timingSafeEqual} from 'node:crypto';export {FiniteQaBudget} from './ops/private-diagnostic-qa/budget.mjs';
const reply=s=>new Response(null,{status:s,headers:{'Cache-Control':'private, no-store','X-Robots-Tag':'noindex, nofollow'}});
export default {async fetch(r,e){try{const u=new URL(r.url);if(u.origin!==e.QA_PREVIEW_ORIGIN||u.pathname!=='/api/private-qa/run'||u.search||r.method!=='POST')return reply(404);if(!qaAccessWindow(e)||e.PRIVATE_QA_MODE!=='finite-synthetic-v1')return reply(503);if(r.headers.get('origin')!==e.QA_PREVIEW_ORIGIN)return reply(403);const raw=r.headers.get('authorization');if(!raw||!/^Basic [A-Za-z0-9+/]+={0,2}$/.test(raw)||raw.length>256)return reply(401);const decoded=atob(raw.slice(6));if(!decoded.startsWith('qa:'))return reply(401);const actual=await qaAccessVerifier(decoded.slice(3));if(!timingSafeEqual(Buffer.from(actual,'hex'),Buffer.from(e.QA_ACCESS_SHA256,'hex')))return reply(401);if(r.body){const rd=r.body.getReader(),p=await rd.read();await rd.cancel();if(!p.done)return reply(400);}const guard=e.PRIVATE_QA_BUDGET.get(e.PRIVATE_QA_BUDGET.idFromName('finite-private-qa-once-v1'));return runFixedQa({...e,PRIVATE_QA_GUARD:guard});}catch{return reply(503);}}};
